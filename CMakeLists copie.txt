cmake_minimum_required(VERSION 3.16)

project(CoastWarn VERSION 1.0 LANGUAGES CXX)

#set(CMAKE_CXX_VISIBILITY_PRESET hidden)
#set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
#set(CMAKE_VERBOSE_MAKEFILE ON)
#set(CMAKE_AUTOMOC ON)
#set(CMAKE_AUTORCC ON)
#set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)

find_package(Qt6 REQUIRED COMPONENTS Quick Positioning)

qt_standard_project_setup(REQUIRES 6.10)

if(ANDROID)
    message("Building for Android")
elseif(APPLE)
    if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
        # Resources for MacOS
        message("Building for MacOS")

        #set(MACOSX_BUNDLE_ICON_FILE CoastWarn.icns)
        set(app_icon "${CMAKE_SOURCE_DIR}/resources/macos/CoastWarn.icns")
        set_source_files_properties(${app_icon} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources")

        set(INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/resources/macos/Info.plist")

        qt_add_executable(CoastWarn
            MACOSX_BUNDLE
            main.cpp
            ${app_icon}
        )

        set_target_properties(CoastWarn PROPERTIES
            MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
            MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
            MACOSX_BUNDLE TRUE
            MACOSX_BUNDLE_INFO_PLIST ${INFO_PLIST}
            MACOSX_BUNDLE_ICON_FILE CoastWarn.icns
        )
    elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        # Resources for iOS
        message("Building for iOS")

        qt_add_executable(CoastWarn
            main.cpp
            ${app_icon}
        )

        set(app_icon "${CMAKE_SOURCE_DIR}/resources/ios/CoastWarn.icns")
        set_source_files_properties(${app_icon} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources")
        set(INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/resources/ios/Info.plist")

        set_target_properties(CoastWarn PROPERTIES
            MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
            MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
            MACOSX_BUNDLE TRUE
            MACOSX_BUNDLE_INFO_PLIST ${INFO_PLIST}
            MACOSX_BUNDLE_ICON_FILE CoastWarn.icns
        )
    elseif(CMAKE_SYSTEM_NAME STREQUAL "")
        message("Building for unknown system")
    endif()
endif()

qt_add_qml_module(CoastWarn
    URI CoastWarn
    QML_FILES
        Main.qml
        SOURCES ShomCoastDownloader.h
        SOURCES ShomCoastDownloader.cpp
        SOURCES CoastDistanceCalculator.h
        SOURCES CoastDistanceCalculator.cpp
        SOURCES AppController.h AppController.cpp
)

target_sources(CoastWarn PRIVATE
    Readme.md
    ${INFO_PLIST}
)

target_link_libraries(CoastWarn
    PRIVATE Qt6::Quick Qt6::Positioning
)

# Packaging configuration
if(APPLE)
    if(IOS AND launch_images)
        # Packaging for iOS
        message("Building for MacOS")
        file(GLOB_RECURSE launch_images CONFIGURE_DEPENDS "resources/ios/LaunchImage*.png")

        target_sources(CoastWarn PRIVATE ${launch_images})
        set_source_files_properties(
            ${launch_images}
            PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    endif()
elseif(ANDROID)
    qt_add_resources(CoastWarn "configuration"
        PREFIX "/"
        FILES
            resources/android/AndroidManifest.xml
            resources/android/gradle.properties
    )
    set_property(TARGET CoastWarn APPEND PROPERTY
        QT_ANDROID_PACKAGE_SOURCE_DIR ${CMAKE_SOURCE_DIR}/resources/android
    )
    include(/Applications/Developpement/Android/android_openssl/CMakeLists.txt)
endif()

install(TARGETS CoastWarn
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)
